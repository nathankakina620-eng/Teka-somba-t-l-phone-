import React, { useState } from "react";
import { Listing } from "../types";
import { 
  Package, 
  Plus, 
  Search, 
  Edit3, 
  Trash2, 
  Eye, 
  DollarSign, 
  Image as ImageIcon, 
  FileText, 
  Check, 
  X, 
  Smartphone, 
  ShieldCheck, 
  AlertCircle,
  Tag,
  ArrowUpDown
} from "lucide-react";
import { useTheme } from "../context/ThemeContext";

interface AdminProductManagerProps {
  listings: Listing[];
  onOpenCreateModal: () => void;
  onOpenEditModal: (listing: Listing) => void;
  onViewDetails: (listing: Listing) => void;
  onDeleteListing: (listingId: string) => void;
  onQuickUpdatePrice: (listingId: string, newPrice: number) => void;
  onToggleAvailability: (listingId: string, isAvailable: boolean) => void;
}

export default function AdminProductManager({
  listings,
  onOpenCreateModal,
  onOpenEditModal,
  onViewDetails,
  onDeleteListing,
  onQuickUpdatePrice,
  onToggleAvailability,
}: AdminProductManagerProps) {
  const { isLight } = useTheme();

  // Search and filter state
  const [searchQuery, setSearchQuery] = useState("");
  const [statusFilter, setStatusFilter] = useState<"ALL" | "AVAILABLE" | "UNAVAILABLE">("ALL");
  const [selectedBrand, setSelectedBrand] = useState("Tous");

  // Inline price edit state: listingId -> new price value
  const [editingPriceId, setEditingPriceId] = useState<string | null>(null);
  const [inlinePriceInput, setInlinePriceInput] = useState<string>("");

  // Confirmation before delete
  const [confirmDeleteId, setConfirmDeleteId] = useState<string | null>(null);

  // Stats calculation
  const totalListings = listings.length;
  const availableListings = listings.filter(l => l.isAvailable).length;
  const unavailableListings = totalListings - availableListings;
  const totalStockValueUSD = listings.reduce((sum, l) => sum + (l.isAvailable ? l.price : 0), 0);

  const brands = ["Tous", ...Array.from(new Set(listings.map(l => l.phone.brand)))];

  // Filter listings
  const filteredListings = listings.filter((l) => {
    const matchesSearch = 
      l.title.toLowerCase().includes(searchQuery.toLowerCase()) ||
      l.phone.model.toLowerCase().includes(searchQuery.toLowerCase()) ||
      l.phone.brand.toLowerCase().includes(searchQuery.toLowerCase()) ||
      l.phone.imei.toLowerCase().includes(searchQuery.toLowerCase());

    const matchesStatus = 
      statusFilter === "ALL" || 
      (statusFilter === "AVAILABLE" && l.isAvailable) ||
      (statusFilter === "UNAVAILABLE" && !l.isAvailable);

    const matchesBrand = selectedBrand === "Tous" || l.phone.brand === selectedBrand;

    return matchesSearch && matchesStatus && matchesBrand;
  });

  const handleStartInlinePriceEdit = (listing: Listing) => {
    setEditingPriceId(listing.id);
    setInlinePriceInput(listing.price.toString());
  };

  const handleSaveInlinePrice = (listingId: string) => {
    const parsed = parseFloat(inlinePriceInput);
    if (!isNaN(parsed) && parsed > 0) {
      onQuickUpdatePrice(listingId, parsed);
    }
    setEditingPriceId(null);
    setInlinePriceInput("");
  };

  return (
    <div className="space-y-6">
      
      {/* Top Banner & Quick Stats */}
      <div className={`p-6 rounded-3xl border shadow-sm transition-colors ${
        isLight 
          ? "bg-white border-amber-500/20" 
          : "bg-gradient-to-br from-neutral-900 via-neutral-900 to-neutral-950 border-amber-500/10"
      }`}>
        <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-4">
          <div className="flex items-center gap-3">
            <div className="p-2.5 bg-amber-400 text-neutral-950 rounded-2xl shadow-xs font-black">
              <Package className="w-6 h-6" />
            </div>
            <div>
              <div className="flex items-center gap-2">
                <span className="text-[10px] font-black uppercase font-mono px-2 py-0.5 rounded bg-amber-500/10 text-amber-600 dark:text-amber-400 border border-amber-500/20">
                  CATALOGUE & ARTICLES
                </span>
                <span className="text-[10px] font-black uppercase font-mono px-2 py-0.5 rounded bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 border border-emerald-500/20">
                  ADMINISTRATEUR ACTIF
                </span>
              </div>
              <h2 className={`text-base sm:text-lg font-black tracking-tight font-display mt-0.5 ${
                isLight ? "text-slate-900" : "text-white"
              }`}>
                Gestion des Produits, Prix & Photos
              </h2>
              <p className={`text-xs ${isLight ? "text-slate-500" : "text-neutral-400"}`}>
                Gérez en direct les photos, les prix en USD/CDF et les descriptions de tous les téléphones du catalogue.
              </p>
            </div>
          </div>

          {/* Add product button */}
          <button
            onClick={onOpenCreateModal}
            className="bg-amber-400 hover:bg-amber-500 text-neutral-950 font-black text-xs uppercase tracking-wide py-3 px-5 rounded-2xl shadow-md transition-all cursor-pointer flex items-center justify-center gap-2 shrink-0 self-start sm:self-center"
          >
            <Plus className="w-4 h-4 stroke-[3]" />
            Ajouter un Article (Admin)
          </button>
        </div>

        {/* 4 Stats metrics */}
        <div className="grid grid-cols-2 sm:grid-cols-4 gap-3 pt-5 text-center">
          <div className={`p-3.5 rounded-2xl border ${
            isLight ? "bg-slate-50 border-slate-200" : "bg-neutral-950 border-neutral-850"
          }`}>
            <span className={`text-[9px] block uppercase font-bold ${isLight ? "text-slate-500" : "text-neutral-500"}`}>
              Total Articles
            </span>
            <span className={`text-base font-black font-mono ${isLight ? "text-slate-900" : "text-white"}`}>
              {totalListings}
            </span>
          </div>

          <div className={`p-3.5 rounded-2xl border ${
            isLight ? "bg-slate-50 border-slate-200" : "bg-neutral-950 border-neutral-850"
          }`}>
            <span className={`text-[9px] block uppercase font-bold ${isLight ? "text-slate-500" : "text-neutral-500"}`}>
              En Vente (Disponibles)
            </span>
            <span className="text-base font-black text-emerald-500 font-mono">
              {availableListings}
            </span>
          </div>

          <div className={`p-3.5 rounded-2xl border ${
            isLight ? "bg-slate-50 border-slate-200" : "bg-neutral-950 border-neutral-850"
          }`}>
            <span className={`text-[9px] block uppercase font-bold ${isLight ? "text-slate-500" : "text-neutral-500"}`}>
              Vendus / Masqués
            </span>
            <span className="text-base font-black text-neutral-400 font-mono">
              {unavailableListings}
            </span>
          </div>

          <div className={`p-3.5 rounded-2xl border ${
            isLight ? "bg-slate-50 border-slate-200" : "bg-neutral-950 border-neutral-850"
          }`}>
            <span className={`text-[9px] block uppercase font-bold ${isLight ? "text-slate-500" : "text-neutral-500"}`}>
              Valeur Stock ($ USD)
            </span>
            <span className="text-base font-black text-amber-500 font-mono">
              ${totalStockValueUSD.toLocaleString()}
            </span>
          </div>
        </div>
      </div>

      {/* Search, Status & Brand Filters */}
      <div className={`p-4 rounded-3xl border shadow-sm space-y-3 transition-colors ${
        isLight ? "bg-white border-slate-200" : "bg-neutral-900/60 border-neutral-850"
      }`}>
        <div className="flex flex-col sm:flex-row gap-3">
          {/* Search */}
          <div className="relative flex-1">
            <Search className={`w-4 h-4 absolute left-3.5 top-3 ${isLight ? "text-slate-400" : "text-neutral-500"}`} />
            <input
              type="text"
              placeholder="Rechercher par titre, modèle, marque ou numéro IMEI..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              className={`w-full rounded-2xl py-2.5 pl-10 pr-4 text-xs font-semibold focus:outline-none transition-colors border ${
                isLight 
                  ? "bg-slate-50 border-slate-200 text-slate-900 placeholder:text-slate-400 focus:bg-white focus:border-amber-500" 
                  : "bg-neutral-950 border-neutral-850 text-neutral-200 focus:border-amber-500"
              }`}
            />
          </div>

          {/* Status selector */}
          <div className="flex items-center gap-1.5 shrink-0">
            <button
              onClick={() => setStatusFilter("ALL")}
              className={`px-3 py-2 rounded-xl text-[10px] font-black uppercase transition-colors cursor-pointer border ${
                statusFilter === "ALL"
                  ? "bg-amber-400 border-amber-400 text-neutral-950"
                  : isLight ? "bg-slate-100 border-slate-200 text-slate-700" : "bg-neutral-950 border-neutral-850 text-neutral-400"
              }`}
            >
              Tous ({totalListings})
            </button>
            <button
              onClick={() => setStatusFilter("AVAILABLE")}
              className={`px-3 py-2 rounded-xl text-[10px] font-black uppercase transition-colors cursor-pointer border ${
                statusFilter === "AVAILABLE"
                  ? "bg-emerald-500 border-emerald-500 text-neutral-950"
                  : isLight ? "bg-slate-100 border-slate-200 text-slate-700" : "bg-neutral-950 border-neutral-850 text-neutral-400"
              }`}
            >
              En vente ({availableListings})
            </button>
            <button
              onClick={() => setStatusFilter("UNAVAILABLE")}
              className={`px-3 py-2 rounded-xl text-[10px] font-black uppercase transition-colors cursor-pointer border ${
                statusFilter === "UNAVAILABLE"
                  ? "bg-neutral-700 border-neutral-700 text-white"
                  : isLight ? "bg-slate-100 border-slate-200 text-slate-700" : "bg-neutral-950 border-neutral-850 text-neutral-400"
              }`}
            >
              Vendus / Masqués ({unavailableListings})
            </button>
          </div>
        </div>

        {/* Brands selector pills */}
        <div className="flex items-center gap-1.5 overflow-x-auto py-1 scrollbar-none">
          <span className={`text-[10px] font-black uppercase tracking-widest mr-1 ${isLight ? "text-slate-500" : "text-neutral-500"}`}>
            Marque :
          </span>
          {brands.map(brand => (
            <button
              key={brand}
              onClick={() => setSelectedBrand(brand)}
              className={`py-1 px-3 rounded-full text-[10px] font-bold transition-all border cursor-pointer ${
                selectedBrand === brand
                  ? "bg-amber-400 border-amber-400 text-neutral-950 font-extrabold"
                  : isLight
                    ? "bg-slate-100 border-slate-200 text-slate-700 hover:bg-slate-200"
                    : "bg-neutral-950 border-neutral-850 text-neutral-400 hover:bg-neutral-900"
              }`}
            >
              {brand}
            </button>
          ))}
        </div>
      </div>

      {/* Articles Management Table / List */}
      <div className={`border rounded-3xl overflow-hidden shadow-sm transition-colors ${
        isLight ? "bg-white border-slate-200" : "bg-neutral-900/60 border-neutral-850"
      }`}>
        <div className={`p-4 border-b flex items-center justify-between ${
          isLight ? "bg-slate-50 border-slate-200" : "bg-neutral-950 border-neutral-850"
        }`}>
          <div>
            <h3 className={`text-xs font-black uppercase font-display ${isLight ? "text-slate-900" : "text-white"}`}>
              Inventaire des Articles & Prix
            </h3>
            <p className={`text-[9px] font-semibold font-mono ${isLight ? "text-slate-500" : "text-neutral-500"}`}>
              {filteredListings.length} article{filteredListings.length > 1 ? "s" : ""} affiché{filteredListings.length > 1 ? "s" : ""}
            </p>
          </div>

          <span className="text-[10px] font-bold text-amber-500 flex items-center gap-1">
            <Tag className="w-3.5 h-3.5" />
            Édition instantanée activée
          </span>
        </div>

        {filteredListings.length > 0 ? (
          <div className={`divide-y ${isLight ? "divide-slate-200" : "divide-neutral-850"}`}>
            {filteredListings.map((listing) => {
              const isEditingPrice = editingPriceId === listing.id;
              const isConfirmingDelete = confirmDeleteId === listing.id;
              const cdfPrice = Math.round(listing.price * (listing.exchangeRate || 2820)).toLocaleString();

              return (
                <div
                  key={listing.id}
                  className={`p-5 flex flex-col lg:flex-row lg:items-center justify-between gap-5 transition-all ${
                    isLight 
                      ? "hover:bg-slate-50/80" 
                      : "hover:bg-neutral-900 bg-gradient-to-r from-neutral-950/20 via-transparent to-transparent"
                  }`}
                >
                  
                  {/* Left Column: Photo Thumbnail + Product Info + Description */}
                  <div className="flex items-start gap-4 flex-1">
                    
                    {/* Photos Preview Thumbnail */}
                    <div className="relative shrink-0 w-24 h-24 rounded-2xl overflow-hidden border border-slate-200 dark:border-neutral-800 shadow-xs group">
                      <img
                        src={listing.photos[0]}
                        alt={listing.title}
                        className="w-full h-full object-cover group-hover:scale-105 transition-transform"
                        referrerPolicy="no-referrer"
                      />
                      <span className="absolute bottom-1 right-1 bg-black/75 backdrop-blur-xs text-white text-[8px] font-mono font-bold px-1.5 py-0.5 rounded flex items-center gap-1">
                        <ImageIcon className="w-2.5 h-2.5 text-amber-400" />
                        {listing.photos.length}
                      </span>
                    </div>

                    {/* Product Details */}
                    <div className="space-y-1.5 flex-1 min-w-0">
                      <div className="flex items-center gap-2 flex-wrap">
                        <span className={`text-[9px] px-2 py-0.5 rounded font-black uppercase font-mono ${
                          listing.phone.brand === "Apple"
                            ? "bg-neutral-800 text-white"
                            : listing.phone.brand === "Samsung"
                              ? "bg-blue-600 text-white"
                              : "bg-amber-500 text-neutral-950"
                        }`}>
                          {listing.phone.brand}
                        </span>

                        <h4 className={`text-xs sm:text-sm font-black truncate ${
                          isLight ? "text-slate-900" : "text-white"
                        }`}>
                          {listing.title}
                        </h4>

                        <span className={`text-[9px] font-mono px-2 py-0.5 rounded border ${
                          isLight ? "bg-slate-100 text-slate-600 border-slate-200" : "bg-neutral-950 text-neutral-400 border-neutral-800"
                        }`}>
                          {listing.phone.storage}GB / {listing.phone.ram}GB RAM
                        </span>

                        {/* Availability badge */}
                        <span className={`text-[9px] font-black uppercase px-2 py-0.5 rounded ${
                          listing.isAvailable
                            ? "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 border border-emerald-500/20"
                            : "bg-neutral-500/10 text-neutral-400 border border-neutral-500/20"
                        }`}>
                          {listing.isAvailable ? "🟢 En vente" : "🔴 Masqué"}
                        </span>
                      </div>

                      {/* Description Preview (Highlighted request) */}
                      <div className={`p-2 rounded-xl text-[11px] leading-relaxed border ${
                        isLight ? "bg-slate-50 border-slate-200 text-slate-700" : "bg-neutral-950 border-neutral-850 text-neutral-350"
                      }`}>
                        <div className="flex items-start gap-1.5">
                          <FileText className="w-3.5 h-3.5 text-amber-500 shrink-0 mt-0.5" />
                          <p className="line-clamp-2 italic">
                            "{listing.description}"
                          </p>
                        </div>
                      </div>

                      {/* Technical Specs & Seller Info */}
                      <div className="flex items-center gap-3 text-[10px] flex-wrap font-medium">
                        <span className={isLight ? "text-slate-500" : "text-neutral-400"}>
                          IMEI : <code className={`font-mono font-bold ${isLight ? "text-slate-900" : "text-neutral-200"}`}>{listing.phone.imei}</code>
                        </span>
                        <span className={isLight ? "text-slate-500" : "text-neutral-400"}>
                          Batterie : <strong className={isLight ? "text-slate-900" : "text-neutral-200"}>{listing.phone.batteryHealth}%</strong>
                        </span>
                        <span className={isLight ? "text-slate-500" : "text-neutral-400"}>
                          État : <strong className={isLight ? "text-slate-900" : "text-neutral-200"}>{listing.phone.screenCondition}</strong>
                        </span>
                        <span className={isLight ? "text-slate-500" : "text-neutral-400"}>
                          Vendeur : <strong className={isLight ? "text-slate-900" : "text-neutral-200"}>{listing.sellerName}</strong> ({listing.sellerPhone})
                        </span>
                      </div>

                    </div>
                  </div>

                  {/* Middle: PRICE SECTION (Inline quick edit) */}
                  <div className={`p-3.5 rounded-2xl border flex flex-col justify-center items-center min-w-[200px] shrink-0 ${
                    isLight ? "bg-amber-50/50 border-amber-200/80" : "bg-neutral-950 border-neutral-850"
                  }`}>
                    <span className={`text-[9px] uppercase font-black tracking-wider ${isLight ? "text-amber-700" : "text-amber-500"}`}>
                      Prix de Vente
                    </span>

                    {isEditingPrice ? (
                      /* Inline Price Editing Form */
                      <div className="mt-1 flex items-center gap-1.5 w-full">
                        <span className="font-bold text-xs text-amber-500">$</span>
                        <input
                          type="number"
                          min="1"
                          value={inlinePriceInput}
                          onChange={(e) => setInlinePriceInput(e.target.value)}
                          onKeyDown={(e) => {
                            if (e.key === "Enter") handleSaveInlinePrice(listing.id);
                            if (e.key === "Escape") setEditingPriceId(null);
                          }}
                          className={`w-24 rounded-lg px-2 py-1 text-xs font-black font-mono border focus:outline-none focus:border-amber-500 ${
                            isLight ? "bg-white border-slate-300 text-slate-900" : "bg-neutral-900 border-neutral-700 text-white"
                          }`}
                          autoFocus
                        />
                        <button
                          onClick={() => handleSaveInlinePrice(listing.id)}
                          className="p-1 bg-emerald-500 hover:bg-emerald-600 text-black rounded-lg cursor-pointer transition-colors shadow-xs"
                          title="Valider le nouveau prix"
                        >
                          <Check className="w-3.5 h-3.5 stroke-[3]" />
                        </button>
                        <button
                          onClick={() => setEditingPriceId(null)}
                          className="p-1 bg-slate-300 dark:bg-neutral-800 text-neutral-700 dark:text-neutral-300 rounded-lg cursor-pointer transition-colors"
                          title="Annuler"
                        >
                          <X className="w-3.5 h-3.5" />
                        </button>
                      </div>
                    ) : (
                      /* Display price with quick edit pencil */
                      <div className="flex items-center gap-2 mt-1">
                        <span className={`text-lg font-black font-mono ${isLight ? "text-slate-900" : "text-white"}`}>
                          ${listing.price}
                        </span>
                        <button
                          onClick={() => handleStartInlinePriceEdit(listing)}
                          className={`p-1 rounded-lg transition-colors cursor-pointer ${
                            isLight 
                              ? "hover:bg-amber-100 text-amber-700" 
                              : "hover:bg-neutral-900 text-amber-400"
                          }`}
                          title="Modifier le prix rapidement"
                        >
                          <Edit3 className="w-3.5 h-3.5" />
                        </button>
                      </div>
                    )}

                    <span className={`text-[10px] font-mono font-bold mt-0.5 ${isLight ? "text-slate-500" : "text-neutral-400"}`}>
                      ≈ {cdfPrice} CDF
                    </span>
                  </div>

                  {/* Right Column: Actions (Edit Photos, Price & Description, Toggle status, Delete) */}
                  <div className="flex flex-row lg:flex-col gap-2 shrink-0 self-stretch justify-center">
                    
                    {/* Full Edit Modal Trigger (Photos, Price, Description) */}
                    <button
                      onClick={() => onOpenEditModal(listing)}
                      className="bg-amber-400 hover:bg-amber-500 text-neutral-950 font-black text-[11px] uppercase tracking-wide py-2 px-3.5 rounded-xl transition-all cursor-pointer shadow-xs flex items-center justify-center gap-1.5"
                    >
                      <Edit3 className="w-3.5 h-3.5" />
                      Modifier l'article
                    </button>

                    {/* Toggle Availability Button */}
                    <button
                      onClick={() => onToggleAvailability(listing.id, !listing.isAvailable)}
                      className={`text-[10px] font-bold py-1.5 px-3 rounded-xl border transition-colors cursor-pointer flex items-center justify-center gap-1 ${
                        listing.isAvailable
                          ? isLight 
                            ? "bg-slate-100 hover:bg-slate-200 text-slate-700 border-slate-200" 
                            : "bg-neutral-950 hover:bg-neutral-900 text-neutral-400 border-neutral-800"
                          : "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400 border-emerald-500/30"
                      }`}
                    >
                      {listing.isAvailable ? "Masquer du catalogue" : "Remettre en vente"}
                    </button>

                    {/* View Details modal */}
                    <button
                      onClick={() => onViewDetails(listing)}
                      className={`text-[10px] font-bold py-1.5 px-3 rounded-xl border transition-colors cursor-pointer flex items-center justify-center gap-1 ${
                        isLight 
                          ? "bg-white hover:bg-slate-100 text-slate-600 border-slate-200" 
                          : "bg-neutral-950 hover:bg-neutral-900 text-neutral-400 border-neutral-800"
                      }`}
                    >
                      <Eye className="w-3.5 h-3.5" />
                      Aperçu client
                    </button>

                    {/* Delete Article */}
                    {isConfirmingDelete ? (
                      <div className="flex items-center gap-1">
                        <button
                          onClick={() => {
                            onDeleteListing(listing.id);
                            setConfirmDeleteId(null);
                          }}
                          className="bg-red-600 hover:bg-red-700 text-white font-black text-[10px] py-1.5 px-2 rounded-lg cursor-pointer flex-1"
                        >
                          Confirmer ?
                        </button>
                        <button
                          onClick={() => setConfirmDeleteId(null)}
                          className="bg-slate-200 dark:bg-neutral-800 text-xs py-1 px-1.5 rounded-lg cursor-pointer"
                        >
                          <X className="w-3 h-3" />
                        </button>
                      </div>
                    ) : (
                      <button
                        onClick={() => setConfirmDeleteId(listing.id)}
                        className={`text-[10px] font-bold py-1.5 px-3 rounded-xl border text-red-500 transition-colors cursor-pointer flex items-center justify-center gap-1 ${
                          isLight 
                            ? "hover:bg-red-50 border-red-200" 
                            : "hover:bg-red-950/20 border-red-900/30"
                        }`}
                      >
                        <Trash2 className="w-3.5 h-3.5" />
                        Supprimer
                      </button>
                    )}

                  </div>

                </div>
              );
            })}
          </div>
        ) : (
          <div className={`p-12 text-center text-xs space-y-2 ${isLight ? "text-slate-500" : "text-neutral-500"}`}>
            <Package className="w-8 h-8 mx-auto opacity-40" />
            <p className="font-bold">Aucun article ne correspond à vos filtres.</p>
            <p className="text-[11px]">Modifiez la recherche ou cliquez sur "+ Ajouter un Article (Admin)".</p>
          </div>
        )}

      </div>
    </div>
  );
}
